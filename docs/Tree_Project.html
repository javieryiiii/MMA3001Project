 39def scan_images(dataset_dir):
 40	"""Index recognized PNG files in the flat dataset directory.
 41
 42	Args:
 43		dataset_dir: Directory containing modality-tagged PNG files.
 44
 45	Returns:
 46		A mapping from ``(tree_id, season, capture_tag)`` to modality/path
 47		mappings. Filenames that do not match ``IMAGE_PATTERN`` are ignored.
 48	"""
 49	images = {}
 50	for path in sorted(dataset_dir.glob("*.png")):
 51		match = IMAGE_PATTERN.match(path.name)
 52		if match is None:
 53			continue
 54
 55		parts = match.groupdict()
 56		key = (parts["tree_id"], parts["season"], parts["tag"])
 57		images.setdefault(key, {})[parts["modality"]] = path
 58	return images

290def prepare_rgb_gt_dataset(summer_pairs):
291	"""Prepare matched Summer RGB/GT samples using the established RGB transform.
292
293	RGB files must be 480x640 and GT masks must be binary 256x256 PNGs with
294	values 0 or 255. RGB is converted from BGR to RGB, cropped with columns
295	80:560, resized to 256x256 using ``INTER_AREA``, and divided by 255. GT
296	geometry is unchanged and values are converted to 0/1.
297
298	Args:
299		summer_pairs: Mapping from sample keys to modality/path mappings with RGB
300			and GT entries.
301
302	Returns:
303		A tuple ``(inputs, targets, tree_ids, sample_keys)``. Inputs have shape
304		``(N, 256, 256, 3)`` and targets ``(N, 256, 256, 1)``, both float32.
305		RGB values are in [0, 1]; targets are binary. IDs and keys follow sorted
306		sample-key order.
307
308	Raises:
309		ValueError: If source RGB/GT dimensions or GT values violate the expected
310			input format.
311	"""
312	sample_keys = sorted(summer_pairs)
313	inputs = np.empty((len(sample_keys), 256, 256, 3), dtype=np.float32)
314	targets = np.empty((len(sample_keys), 256, 256, 1), dtype=np.float32)
315	tree_ids = []
316
317	for index, key in enumerate(sample_keys):
318		modalities = summer_pairs[key]
319		rgb_bgr = load_image(modalities["RGB"], cv2.IMREAD_COLOR)
320		gt = load_image(modalities["GT"], cv2.IMREAD_GRAYSCALE)
321		if rgb_bgr.shape[:2] != (480, 640):
322			raise ValueError(f"Unexpected RGB dimensions for {modalities['RGB'].name}: {rgb_bgr.shape}")
323		if gt.shape != TARGET_SIZE:
324			raise ValueError(f"Unexpected GT dimensions for {modalities['GT'].name}: {gt.shape}")
325		if not np.isin(np.unique(gt), [0, 255]).all():
326			raise ValueError(f"GT mask is not binary (0/255): {modalities['GT'].name}")
327
328		rgb = cv2.cvtColor(rgb_bgr, cv2.COLOR_BGR2RGB)
329		centre_crop = rgb[:, 80:560]
330		resized_rgb = cv2.resize(centre_crop, TARGET_SIZE, interpolation=cv2.INTER_AREA)
331		inputs[index] = resized_rgb.astype(np.float32) / 255.0
332		targets[index, :, :, 0] = (gt == 255).astype(np.float32)
333		tree_ids.append(key[0])
334
335	return inputs, targets, np.array(tree_ids), sample_keys

338def split_by_tree(tree_ids, sample_keys, seed=RANDOM_SEED):
339	"""Assign samples to deterministic splits while keeping each tree intact.
340
341	Unique tree IDs are sorted by ``np.unique`` and permuted by a NumPy generator
342	initialized with ``seed``. The first floor(70%) of trees are training, the
343	next floor(15%) are validation, and the remainder are test. Sample order
344	within each returned index/key list follows the input order.
345
346	Args:
347		tree_ids: One tree ID per sample.
348		sample_keys: One corresponding sample key per sample.
349		seed: Seed used to permute the unique tree IDs.
350
351	Returns:
352		A mapping for ``train``, ``validation``, and ``test``. Each entry contains
353		an int32 index array, a set of assigned tree IDs, and the corresponding
354		sample keys.
355	"""
356	unique_tree_ids = np.unique(tree_ids)
357	shuffled_tree_ids = np.random.default_rng(seed).permutation(unique_tree_ids)
358	train_tree_count = int(len(unique_tree_ids) * 0.70)
359	validation_tree_count = int(len(unique_tree_ids) * 0.15)
360
361	train_trees = set(shuffled_tree_ids[:train_tree_count])
362	validation_trees = set(
363		shuffled_tree_ids[train_tree_count:train_tree_count + validation_tree_count]
364	)
365	test_trees = set(shuffled_tree_ids[train_tree_count + validation_tree_count:])
366	assignments = {
367		"train": train_trees,
368		"validation": validation_trees,
369		"test": test_trees,
370	}
371
372	splits = {}
373	for split_name, split_trees in assignments.items():
374		indices = np.array(
375			[index for index, tree_id in enumerate(tree_ids) if tree_id in split_trees],
376			dtype=np.int32,
377		)
378		splits[split_name] = {
379			"indices": indices,
380			"tree_ids": split_trees,
381			"sample_keys": [sample_keys[index] for index in indices],
382		}
383
384	return splits

387def verify_tree_split_isolation(splits):
388	"""Check that tree IDs are not assigned to multiple split groups.
389
390	Args:
391		splits: Mapping with ``train``, ``validation``, and ``test`` entries, each
392		containing a ``tree_ids`` set.
393
394	Raises:
395		AssertionError: If any pair of split tree-ID sets overlaps or if the
396			union size differs from the sum of their sizes, indicating duplicate
397			assignment. This check does not verify expected split sizes or missing IDs.
398	"""
399	train_trees = splits["train"]["tree_ids"]
400	validation_trees = splits["validation"]["tree_ids"]
401	test_trees = splits["test"]["tree_ids"]
402	if train_trees & validation_trees or train_trees & test_trees or validation_trees & test_trees:
403		raise AssertionError("Tree identity overlap detected between dataset splits.")
404	all_split_trees = train_trees | validation_trees | test_trees
405	if len(all_split_trees) != sum(len(splits[name]["tree_ids"]) for name in splits):
406		raise AssertionError("Tree IDs are duplicated across dataset splits.")

445def convolution_block(input_tensor, filters):
446	"""Apply two same-padded 3x3 ReLU convolutions.
447
448	Args:
449		input_tensor: Keras feature tensor.
450		filters: Number of output channels for each convolution.
451
452	Returns:
453		Feature tensor with the same spatial dimensions and ``filters`` channels.
454	"""
455	output = tf.keras.layers.Conv2D(
456		filters, 3, activation="relu", padding="same", kernel_initializer="he_normal"
457	)(input_tensor)
458	output = tf.keras.layers.Conv2D(
459		filters, 3, activation="relu", padding="same", kernel_initializer="he_normal"
460	)(output)
461	return output

464def build_unet(input_shape=(256, 256, 3), base_filters=16):
465	"""Build and compile the project U-Net with skip-connected encoder/decoder stages.
466
467	Args:
468		input_shape: Height, width, and channel count. The project RGB models use
469			``(256, 256, 3)``; the RGB-D model supplies four channels.
470		base_filters: Width of the first encoder stage; subsequent stages scale
471			this value by powers of two.
472		Input height and width must be divisible by 16 for the four pooling stages
473		and skip connections to align.
474
475	Returns:
476		A compiled Keras model with a one-channel sigmoid probability output of
477		the same spatial size as the input.
478	"""
479	inputs = tf.keras.Input(shape=input_shape)
480	filters = [base_filters, base_filters * 2, base_filters * 4, base_filters * 8]
481	encoder_features = []
482	output = inputs
483
484	# The encoder pools spatial dimensions while increasing feature channels.
485	for filter_count in filters:
486		output = convolution_block(output, filter_count)
487		encoder_features.append(output)
488		output = tf.keras.layers.MaxPooling2D(pool_size=(2, 2))(output)
489
490	# The bottleneck captures the most compressed, high-level representation.
491	output = convolution_block(output, base_filters * 16)
492	# Upsampling plus skip connections recover detail lost during pooling.
493	for filter_count, skip_features in zip(reversed(filters), reversed(encoder_features)):
494		output = tf.keras.layers.Conv2DTranspose(
495			filter_count, 2, strides=2, padding="same"
496		)(output)
497		output = tf.keras.layers.Concatenate()([output, skip_features])
498		output = convolution_block(output, filter_count)
499
500	outputs = tf.keras.layers.Conv2D(1, 1, activation="sigmoid", name="skeleton_probability")(output)
501	model = tf.keras.Model(inputs=inputs, outputs=outputs, name="rgb_baseline_unet")
502	model.compile(
503		optimizer=tf.keras.optimizers.Adam(learning_rate=1e-3),
504		# Keep BCE easy to replace with Dice or a combined loss in a later stage.
505		loss=LOSS_FUNCTION,
506		metrics=[dice_coefficient, intersection_over_union],
507	)
508	return model

511def dice_coefficient(y_true, y_pred):
512	"""Compute mean soft Dice over a batch of segmentation masks.
513
514	Args:
515		y_true: Ground-truth tensor shaped ``(N, H, W, C)``.
516		y_pred: Soft prediction probabilities with the same shape.
517
518	Returns:
519		Scalar batch mean of per-image Dice values, using ``1e-6`` smoothing in
520		the numerator and denominator.
521	"""
522	y_true = tf.cast(y_true, tf.float32)
523	y_pred = tf.cast(y_pred, tf.float32)
524	intersection = tf.reduce_sum(y_true * y_pred, axis=(1, 2, 3))
525	mask_sum = tf.reduce_sum(y_true + y_pred, axis=(1, 2, 3))
526	dice = (2.0 * intersection + 1e-6) / (mask_sum + 1e-6)
527	return tf.reduce_mean(dice)

530def intersection_over_union(y_true, y_pred):
531	"""Compute mean soft intersection over union (IoU) over a batch.
532
533	Args:
534		y_true: Ground-truth tensor shaped ``(N, H, W, C)``.
535		y_pred: Soft prediction probabilities with the same shape.
536
537	Returns:
538		Scalar batch mean of per-image IoU values, using ``1e-6`` smoothing in
539		the numerator and denominator.
540	"""
541	y_true = tf.cast(y_true, tf.float32)
542	y_pred = tf.cast(y_pred, tf.float32)
543	intersection = tf.reduce_sum(y_true * y_pred, axis=(1, 2, 3))
544	union = tf.reduce_sum(y_true + y_pred - y_true * y_pred, axis=(1, 2, 3))
545	iou = (intersection + 1e-6) / (union + 1e-6)
546	return tf.reduce_mean(iou)

549def binary_dice_coefficient(y_true, y_pred_binary):
550	"""Calculate mean per-image binary Dice for a batch of masks.
551
552	Ground truth is binarized with ``>= 0.5`` and predictions are converted to
553	boolean. All non-batch axes are reduced independently per image.
554
555	Args:
556		y_true: Ground-truth array with shape ``(N, ...)``.
557		y_pred_binary: Binary prediction array with the same shape.
558
559	Returns:
560		Python float containing the mean per-image Dice; ``1e-6`` smoothing keeps
561		empty-mask comparisons defined.
562
563	Raises:
564		ValueError: If the input shapes differ.
565	"""
566	y_true_binary = np.asarray(y_true) >= 0.5
567	y_pred_binary = np.asarray(y_pred_binary, dtype=bool)
568	if y_true_binary.shape != y_pred_binary.shape:
569		raise ValueError("Ground-truth and binary prediction shapes must match.")
570	image_axes = tuple(range(1, y_true_binary.ndim))
571	intersection = np.sum(y_true_binary & y_pred_binary, axis=image_axes)
572	foreground_sum = np.sum(y_true_binary, axis=image_axes) + np.sum(y_pred_binary, axis=image_axes)
573	return float(np.mean((2.0 * intersection + 1e-6) / (foreground_sum + 1e-6)))

576def binary_intersection_over_union(y_true, y_pred_binary):
577	"""Calculate mean per-image binary IoU for a batch of masks.
578
579	Ground truth is binarized with ``>= 0.5`` and predictions are converted to
580	boolean. All non-batch axes are reduced independently per image.
581
582	Args:
583		y_true: Ground-truth array with shape ``(N, ...)``.
584		y_pred_binary: Binary prediction array with the same shape.
585
586	Returns:
587		Python float containing the mean per-image IoU, with ``1e-6`` smoothing.
588
589	Raises:
590		ValueError: If the input shapes differ.
591	"""
592	y_true_binary = np.asarray(y_true) >= 0.5
593	y_pred_binary = np.asarray(y_pred_binary, dtype=bool)
594	if y_true_binary.shape != y_pred_binary.shape:
595		raise ValueError("Ground-truth and binary prediction shapes must match.")
596	image_axes = tuple(range(1, y_true_binary.ndim))
597	intersection = np.sum(y_true_binary & y_pred_binary, axis=image_axes)
598	union = np.sum(y_true_binary | y_pred_binary, axis=image_axes)
599	return float(np.mean((intersection + 1e-6) / (union + 1e-6)))

465def build_unet(input_shape=(256, 256, 3), base_filters=16):
466	"""Build and compile the project U-Net with skip-connected encoder/decoder stages.
467
468	Args:
469		input_shape: Height, width, and channel count. The project RGB models use
470			``(256, 256, 3)``; the RGB-D model supplies four channels.
471		base_filters: Width of the first encoder stage; subsequent stages scale
472			this value by powers of two.
473		Input height and width must be divisible by 16 for the four pooling stages
474		and skip connections to align.
475
476	Returns:
477		A compiled Keras model with a one-channel sigmoid probability output of
478		the same spatial size as the input.
479	"""
480	inputs = tf.keras.Input(shape=input_shape)
481	filters = [base_filters, base_filters * 2, base_filters * 4, base_filters * 8]
482	encoder_features = []
483	output = inputs
484
485	# The encoder pools spatial dimensions while increasing feature channels.
486	for filter_count in filters:
487		output = convolution_block(output, filter_count)
488		encoder_features.append(output)
489		output = tf.keras.layers.MaxPooling2D(pool_size=(2, 2))(output)
490
491	# The bottleneck captures the most compressed, high-level representation.
492	output = convolution_block(output, base_filters * 16)
493	# Upsampling plus skip connections recover detail lost during pooling.
494	for filter_count, skip_features in zip(reversed(filters), reversed(encoder_features)):
495		output = tf.keras.layers.Conv2DTranspose(
496			filter_count, 2, strides=2, padding="same"
497		)(output)
498		output = tf.keras.layers.Concatenate()([output, skip_features])
499		output = convolution_block(output, filter_count)
500
501	outputs = tf.keras.layers.Conv2D(1, 1, activation="sigmoid", name="skeleton_probability")(output)
502	model = tf.keras.Model(inputs=inputs, outputs=outputs, name="rgb_baseline_unet")
503	model.compile(
504		optimizer=tf.keras.optimizers.Adam(learning_rate=1e-3),
505		# Keep BCE easy to replace with Dice or a combined loss in a later stage.
506		loss=LOSS_FUNCTION,
507		metrics=[dice_coefficient, intersection_over_union],
508	)
509	return model
